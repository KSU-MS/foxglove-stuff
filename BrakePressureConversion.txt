// This is for foxglove specific types
import { Input } from "./types";

// This script is for the KS9E. This is used because the conversion for reading to pressure did not get implicated in the raw readout, so an additional script is needed to make it work.

// Set output variables
type Output = {
  front_BP: number;
  rear_BP: number;
  front_bias: number;
};
type GlobalVariables = { id: number };

// Local vars to calc shit
var front_reading = 0;
var rear_reading = 0;
var front_voltage = 0;
var rear_voltage = 0;
var front_pressure = 0;
var rear_pressure = 0;

// List all the input topics in the input array
export const inputs = [
  "cornernode_rear_brakepressure_data",
  "vcu_pedal_readings_data",
];
export const output = "/math_script/brakepressures";

// Make an InputEvent type alias. Since our script will get a message from either input topic, we need to enumerate the topics.
type InputEvent =
  | Input<"cornernode_rear_brakepressure_data">
  | Input<"vcu_pedal_readings_data">;

export default function script(
  event: InputEvent,
  globalVars: GlobalVariables,
): Output {
  switch (event.topic) {
    case "vcu_pedal_readings_data":
      front_reading = event.message.BSE1;
      front_voltage = front_reading * (5 / 4096) - 0.5;
      front_pressure = (front_voltage * 375) + 107.849;
      break;

    case "cornernode_rear_brakepressure_data":
      rear_reading = event.message.cornernode_rear_brakepressure;
      rear_voltage = rear_reading * (3.3 / 4096) * (1.5035 / 1.0008) - 0.5;
      rear_pressure = rear_voltage * 375;
      break;
  }

// The multiplication by 375 is for a factor calculated as the inverse of (4.5 - 0.5) / 1500. I think 4.5 - 0.5 is the voltage range.

  return {
    front_BP: front_pressure,
    rear_BP: rear_pressure,
    front_bias: 100*(front_pressure/(front_pressure + rear_pressure)) // PERCENT
  };
}
